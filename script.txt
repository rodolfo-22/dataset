// =========================================================================
// 1. CARGA Y FILTRADO DE LOS LÍMITES MUNICIPALES (GAUL NIVEL 2)
// =========================================================================
// Invocamos la base de datos oficial de límites político-administrativos de la ONU
var municipiosSantaAna = ee.FeatureCollection("FAO/GAUL/2015/level2")
  .filter(ee.Filter.eq('clean_country_name', 'El Salvador'))
  .filter(ee.Filter.eq('clean_adm1_name', 'Santa Ana'));

// Centramos el mapa en la región de estudio y añadimos una capa de control visual
Map.centerObject(municipiosSantaAna, 10);
Map.addLayer(municipiosSantaAna, {color: 'blue'}, 'Municipios de Santa Ana');

// Justificación Técnica: El uso del asset oficial GAUL 2 garantiza la estandarización cartográfica internacional de las fronteras municipales salvadoreñas, aislando los polígonos del lado del servidor de Google para optimizar la memoria.
// Fuente Científica (APA): FAO. (2015). Global Administrative Unit Layers (GAUL). FAO GeoNetwork.


// =========================================================================
// 2. CARGA Y SEGMENTACIÓN TEMPORAL DE LA COLECCIÓN FAO ASIS (2016-2025)
// =========================================================================
// Cargamos el producto del Índice de Salud de la Vegetación Mensual
var coleccionVHI = ee.ImageCollection("FAO/GIEWS/ASIS/VHI_M");

// Aplicamos el filtro para los años solicitados (2016-2025) y los meses del maíz de primera
var coleccionFiltrada = coleccionVHI
  .filter(ee.Filter.calendarRange(2016, 2025, 'year'))
  .filter(ee.Filter.calendarRange(6, 8, 'month'));

// Justificación Técnica: Acotar la serie temporal de 2016 a 2025 a los meses de junio, julio y agosto alinea el modelo con el calendario de siembra de primera, eliminando el ruido climático de la época seca y de otros ciclos agrícolas.
// Fuente Científica (APA): FEWS NET. (2018). Central America and the Caribbean Crop Calendar. Famine Early Warning Systems Network / USAID.


// =========================================================================
// 3. PROCESAMIENTO MUNICIPAL (ESTADÍSTICAS DE ZONA / MAP-REDUCE)
// =========================================================================
var calcularPromedioMunicipal = function(imagen) {
  // Extraemos la fecha de la imagen en formato Año-Mes para indexar la serie temporal
  var fechaStr = imagen.date().format('YYYY-MM');
  
  // Ejecutamos la reducción zonal (Zonal Statistics) sobre los municipios
  return imagen.reduceRegions({
    collection: municipiosSantaAna,
    reducer: ee.Reducer.mean(),      // Calcula la media aritmética del VHI por municipio
    scale: 1000                      // Resolución espacial nativa del sensor (1 Km)
  }).map(function(feature) {
    // Vinculamos la fecha de registro como un atributo para cada fila
    return feature.set('fecha_registro', fechaStr);
  });
};

// Aplicamos el algoritmo a toda la serie temporal de forma paralela y aplanamos la matriz
var datasetMunicipalCompleto = coleccionFiltrada.map(calcularPromedioMunicipal).flatten();

// Justificación Técnica: El reductor 'ee.Reducer.mean()' ejecuta estadísticas zonales nativas agregando la varianza de los píxeles a nivel de polígono. El método '.flatten()' transforma las estructuras de datos tridimensionales de las imágenes en un formato tabular bidimensional apto para Machine Learning.
// Fuente Científica (APA): Westra, E. (2016). Python Geospatial Analysis Cookbook. Packt Publishing Ltd. (Capítulo 5: Working with Raster Data and Zonal Statistics).

// =========================================================================
// 4. EXPORTACIÓN AUTOMATIZADA A GOOGLE DRIVE
// =========================================================================
Export.table.toDrive({
  collection: datasetMunicipalCompleto,
  description: 'vhi_municipal_santa_ana_2016_2025',
  fileFormat: 'CSV',
  // Especificamos las columnas estrictamente necesarias para limpiar el archivo de geometrías pesadas
  selectors: ['clean_adm2_name', 'clean_adm1_name', 'fecha_registro', 'mean']
});

// Justificación Técnica: El módulo 'Export.table.toDrive' delega una tarea asíncrona por lotes en los clusters de Google. Al definir los 'selectors', se descartan los vértices geográficos pesados, generando un archivo CSV optimizado para su lectura inmediata con la librería Pandas en Python.
// Fuente Científica (APA): Gorelick, N., Hancher, M., Dixon, M., Ilyushchenko, S., Thau, D., & Moore, R. (2017). Google Earth Engine: Planetary-scale geospatial analysis for everyone. Remote Sensing of Environment, 202, 18-27.
